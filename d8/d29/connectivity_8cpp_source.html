70static void progress(void) {
71 int k, n = 0, out = 0, err;
72
73 if (nact == 0) {
74 return;
75 }
76 err = MPI_Testsome(nact, req, &out, idx, st);
77 if (err != MPI_SUCCESS && err != MPI_ERR_IN_STATUS) {
78 /* Outcome per request is unknown; treat all as failed rather than spin. */
79 for (k = 0; k < nact; ++k) {
80 *flag[k] = FAILED;
81 }
82 nact = 0;
83 return;
84 }
85 if (out == MPI_UNDEFINED) {
86 out = 0;
87 }
88 for (k = 0; k < out; ++k) {
89 *flag[idx[k]] = (err == MPI_ERR_IN_STATUS && st[k].MPI_ERROR != MPI_SUCCESS) ? FAILED : DONE;
90 }
91 for (k = 0; k < nact; ++k) {
92 if (*flag[k] == PENDING) {
93 req[n] = req[k];
94 flag[n] = flag[k];
95 ++n;
96 }
97 }
98 nact = n;
99}

101/* Polls until all n requests complete (1), or the deadline passes or an error occurs (0). */
102static int wait_until(MPI_Request *r, int n, double deadline) {
103 int done;
104
105 for (;;) {
106 if (MPI_Testall(n, r, &done, MPI_STATUSES_IGNORE) != MPI_SUCCESS) {
107 return 0;
108 }
109 if (done) {
110 return 1;
111 }
112 if (MPI_Wtime() > deadline) {
113 return 0;
114 }
115 }

118static int next_run(const char *s, int n, int *j, int *lo, int *hi) {
119 while (*j < n && s[*j] == DONE) {
120 ++*j;
121 }
122 if (*j == n) {
123 return 0;
124 }
125 *lo = *j;
126 while (*j < n && s[*j] != DONE) {
127 ++*j;
128 }
129 *hi = *j - 1;

230/* Circle method: with P even, every pair meets exactly once in steps 0..P-2; a partner >= size is a bye. */
231static int partner(int i, int step, int P) {
232 if (i == P - 1) {
233 return step;
234 }
235 if (i == step) {
236 return P - 1;
237 }
238 return ((2 * step - i) % (P - 1) + (P - 1)) % (P - 1);
239}

250/* Sleeps instead of spinning so cores sit idle; loops because nanosleep can
251 * wake early on a signal. */
252static void pause_for(double seconds) {
253 double end = MPI_Wtime() + seconds, left;
254
255 while ((left = end - MPI_Wtime()) > 0) {
256 struct timespec ts;
257
258 ts.tv_sec = (time_t)left;
259 ts.tv_nsec = (long)((left - (double)ts.tv_sec) * 1e9);
260 nanosleep(&ts, NULL);
261 }
262}

264/* One timed round trip per pair per round. Returns the number of stalls; sets
265 * *gave_up if a barrier timed out. A stalled pair is dropped for good, because
266 * its abandoned requests would match later messages. */
267static int latency(int rounds, double pause, float *samp, int *pstep, char *dead, int *inbuf, int *gave_up) {
268 int P = size + (size & 1), r, step, stalls = 0;
269 MPI_Request rq[2];
270
271 *gave_up = 0;
272 for (r = 0; r < rounds; ++r) {
273 /* Spreads rounds over time so intermittent problems get sampled; the next barrier absorbs any skew in waking up. */
274 if (r > 0) {
275 pause_for(pause);
276 }
277 for (step = 0; step < P - 1; ++step) {
278 int p = partner(rank, step, P), ok;
279 double dl, t;
280
281 /* Lockstep caps the delay a stalled pair imposes on everyone else at one timeout, so stalls can't cascade. */
282 if (MPI_Ibarrier(MPI_COMM_WORLD, &rq[0]) != MPI_SUCCESS || !wait_until(rq, 1, MPI_Wtime() + 2 * timeout)) {
283 printf("rank %d (%s): latency barrier timed out in round %d, giving up\n", rank, host, r);
284 fflush(stdout);
285 *gave_up = 1;
286 return stalls;
287 }
288 if (p >= size || dead[p]) {
289 continue;
290 }
291 dl = MPI_Wtime() + timeout;
292 if (rank < p) {
293 /* Untimed first exchange absorbs the partner leaving the barrier later than us. */
294 ok = lat_recv(&inbuf[p], p, &rq[0]) &&
295 lat_send(p, &rq[1]) &&
296 wait_until(rq, 2, dl) &&
297 lat_recv(&inbuf[p], p, &rq[0]);
298 if (ok) {
299 t = MPI_Wtime();
300 ok = lat_send(p, &rq[1]) && wait_until(rq, 2, dl);
301 if (ok) {
302 int i = p - rank - 1;
303
304 /* Indexed by round, and each pair keeps its step every round, which places every spike in time. */
305 samp[(size_t)i * rounds + r] = (float)(MPI_Wtime() - t);
306 pstep[i] = step;
307 }
308 }
309 } else {
310 /* Preposting the timed ping's receive keeps it off the unexpected-message path. */
311 ok = lat_recv(&inbuf[p], p, &rq[0]) &&
312 wait_until(rq, 1, dl) &&
313 lat_recv(&inbuf[p], p, &rq[0]) &&
314 lat_send(p, &rq[1]) &&
315 wait_until(&rq[1], 1, dl) &&
316 wait_until(rq, 1, dl) &&
317 lat_send(p, &rq[1]) &&
318 wait_until(&rq[1], 1, dl);
319 }
320 if (!ok) {
321 dead[p] = 1;
322 ++stalls;
323 printf("rank %d (%s): latency exchange with rank %d stalled in round %d\n", rank, host, p, r);
324 fflush(stdout);
325 }
326 }
327 }
328 return stalls;
329}

352 double thr;

356 s->b = rank + 1 + i;

388 continue;
389 }

405 free(x);
406 free(all);
407 return n;
408}

598 }
599 ++hc[nhc - 1].n;
600 }
601 qsort(hc, nhc, sizeof *hc, by_count_desc);
602 if (gn == 1) {
603 printf(" round %d, step %d: 1 spike, %.2f us:", ev[g0].round, ev[g0].step, 1e6 * mn);
604 } else {
605 printf(" round %d, step %d: %ld spikes, %.2f to %.2f us:", ev[g0].round, ev[g0].step, gn, 1e6 * mn, 1e6 * gk[go[k]]);
606 }
607 for (q = 0; q < nhc && q < TOPPAIRS; ++q) {
608 printf("%s %s - %s (%ld)", q ? "," : "", hname[hc[q].hp / nh], hname[hc[q].hp % nh], hc[q].n);
609 }
610 if (nhc > TOPPAIRS) {
611 printf(", +%ld more host pair(s)", nhc - TOPPAIRS);
612 }
613 printf("\n");
614 }
615 if (ntrunc > 0) {
616 printf(" each rank reports at most its %d largest spikes and %d rank(s) had more, so these counts are lower bounds\n", MAXEV, ntrunc);
617 }
618 }
619
620 /* A host is flagged when most of its links to other hosts spike: a faulty
621 * NIC raises all of them, while a host merely talking to a faulty one has
622 * a minority of bad links. The lower median makes "most" a strict
623 * majority, so with 3 hosts an innocent host with one bad partner is not
624 * flagged. */
625 nsel = 0;
626 if (nh >= 3) {
627 long nl = 0;
628 double base, mtyp;
629
630 hsp = (long *)calloc((size_t)nh * nh, sizeof *hsp);
631 hsm = (long *)calloc((size_t)nh * nh, sizeof *hsm);
632 rate = (double *)malloc(((size_t)nh * nh + 1) * sizeof *rate);
633 if (!hsp || !hsm || !rate) {
634 fprintf(stderr, "rank 0: out of memory for host statistics\n");
635 return nslow > 0;
636 }
637 for (p = 0; p < nps; ++p) {
638 int ha = hid[ps[p].a], hb = hid[ps[p].b];
639
640 if (ps[p].n == 0 || ha == hb) {
641 continue;
642 }
643 hsp[(size_t)ha * nh + hb] += ps[p].nspike;
644 hsm[(size_t)ha * nh + hb] += ps[p].n;
645 hsp[(size_t)hb * nh + ha] += ps[p].nspike;
646 hsm[(size_t)hb * nh + ha] += ps[p].n;
647 }
648 for (j = 0; j < nh; ++j) {
649 for (k = j + 1; k < nh; ++k) {
650 if (hsm[(size_t)j * nh + k] > 0) {
651 rate[nl++] = (double)hsm[(size_t)j * nh + k];
652 }
653 }
654 }
655 qsort(rate, nl, sizeof *rate, by_double);
656 mtyp = nl > 0 ? rate[nl / 2] : 1.0;
657 for (j = 0; j < nh; ++j) {
658 nl = 0;
659 for (k = 0; k < nh; ++k) {
660 if (k != j && hsm[(size_t)j * nh + k] > 0) {
661 rate[nl++] = (double)hsp[(size_t)j * nh + k] / hsm[(size_t)j * nh + k];
662 }
663 }
664 qsort(rate, nl, sizeof *rate, by_double);
665 key[j] = nl > 0 ? rate[(nl - 1) / 2] : -1.0;
666 }
667 /* Baseline from host scores, not from links: with few hosts, most links can touch one bad host. */
668 nl = 0;
669 for (j = 0; j < nh; ++j) {
670 if (key[j] >= 0) {
671 rate[nl++] = key[j];
672 }
673 }
674 qsort(rate, nl, sizeof *rate, by_double);
675 base = nl > 0 ? rate[(nl - 1) / 2] : 0.0;
676 /* One extra spike per typical link keeps lone spikes on a quiet machine from flagging anything. */
677 thr = factor * (base + 1.0 / mtyp);
678 for (j = 0; j < nh; ++j) {
679 if (key[j] > thr) {
680 sel[nsel++] = j;
681 }
682 }
683 g_key = key;
684 qsort(sel, nsel, sizeof *sel, by_key_desc);
685 printf(" %ld host(s) where most links to other hosts spike > %gx as often as on a typical host%s\n", nsel, factor, nsel ? ":" : "");
686 for (k = 0; k < nsel; ++k) {
687 printf(" %s: most links spike at a rate >= %.3g; typical host %.3g\n", hname[sel[k]], key[sel[k]], base);
688 }
689 }
690 fflush(stdout);
691 return nslow > 0 || nsel > 0;
692}

704/* In enum order. The parser and the help text both come from this table, so they cannot drift apart. */
705/* clang-format off */
706static const struct {
707 const char *name;
708 char letter;
709 const char *metavar, *expect, *help;
710} option_table[] = {
711 {"timeout", 't', "S", "seconds, more than 0 and at most 86400", "seconds until a missing message counts as failed (default 30, at most 86400)"},
712 {"rounds", 'r', "N", "a whole number from 0 to 1000000", "latency rounds after a passed check (default 0: none, at most 1000000)"},
713 {"factor", 'f', "F", "a number more than 1 and at most 1e6", "outlier factor for slow pairs, spikes and hosts (default 10)"},
714 {"pause", 'p', "S", "seconds from 0 to 86400", "seconds between latency rounds (default 10, at most 86400)"},
715 {"machine", 'm', NULL, NULL, "also print one machine-readable line per rank: @connectivity v=1 rank= size= node= host= missing="},
716 {"help", 'h', NULL, NULL, "show this help and exit"},
717};

721static void usage(FILE *f, const char *prog) {
722 int k;
723
724 fprintf(f, "usage: %s [options]\n", prog);
725 fprintf(f, "Checks that every rank receives a message from every other rank, then optionally times round trips between all pairs.\n");
726 for (k = 0; k < NOPT; ++k) {
727 const char *mv = option_table[k].metavar;
728 int pad = 12 - (int)(strlen(option_table[k].name) + (mv ? 1 + strlen(mv) : 0));
729
730 fprintf(f, " -%c, --%s%s%s%*s %s\n", option_table[k].letter, option_table[k].name, mv ? "=" : "", mv ? mv : "", pad, "", option_table[k].help);
731 }

733 fprintf(f, "Exit status: 0 all good, 1 a failed check or incomplete latency rounds, 2 latency outliers, %d bad command line.\n", EXIT_USAGE);

736static int usage_hint(const char *prog) {
737 if (rank == 0) {
738 fprintf(stderr, "Try '%s --help'.\n", prog);
739 }
740 return EXIT_USAGE;
741}

743/* The whole string must be a number in [lo, hi], or (lo, hi] if lo_open. NaN fails every comparison, inf fails the upper bound. */
744static int parse_double(const char *s, double lo, int lo_open, double hi, double *v) {
745 char *end;
746 double x = strtod(s, &end);
747
748 if (end == s || *end != '\0' || !(lo_open ? x > lo : x >= lo) || !(x <= hi)) {
749 return 0;
750 }
751 *v = x;
752 return 1;
753}
754
755static int parse_int(const char *s, long lo, long hi, int *v) {
756 char *end;
757 long x = strtol(s, &end, 10);
758
759 /* strtol saturates on overflow, which the range check then rejects. */
760 if (end == s || *end != '\0' || x < lo || x > hi) {
761 return 0;
762 }
763 *v = (int)x;
764 return 1;
765}

767static int option_index(int letter) {
768 int k;
769
770 for (k = 0; k < NOPT; ++k) {
771 if (option_table[k].letter == letter) {
772 return k;
773 }
774 }
775 return -1;
776}

778/* Returns 0 to go on, -1 after --help, or EXIT_USAGE. Every rank parses the same arguments, so only rank 0 reports. getopt_long is a GNU extension that musl and the BSDs also provide. */
779static int parse_args(int argc, char **argv, options_t *o) {
780 const char *prog = argc > 0 ? argv[0] : "connectivity";
781 struct option longopts[NOPT + 1];
782 char shortopts[2 * NOPT + 1];
783 int c, k, n = 0, ok;
784
785 o->timeout = 30.0;
786 o->rounds = 0;
787 o->factor = 10.0;
788 o->pause = 10.0;
789 o->machine = 0;
790 /* Built from option_table, so getopt_long sees exactly the options the help text describes. */
791 for (k = 0; k < NOPT; ++k) {
792 longopts[k].name = option_table[k].name;
793 longopts[k].has_arg = option_table[k].metavar ? required_argument : no_argument;
794 longopts[k].flag = NULL;
795 longopts[k].val = option_table[k].letter;
796 shortopts[n++] = option_table[k].letter;
797 if (option_table[k].metavar) {
798 shortopts[n++] = ':';
799 }
800 }
801 memset(&longopts[NOPT], 0, sizeof longopts[NOPT]);
802 shortopts[n] = '\0';
803 opterr = rank == 0;
804 while ((c = getopt_long(argc, argv, shortopts, longopts, NULL)) != -1) {
805 k = option_index(c);
806 if (k < 0) {
807 /* getopt_long has already said what was wrong. */
808 return usage_hint(prog);
809 }
810 ok = 1;
811 switch (k) {
812 case OPT_TIMEOUT:
813 ok = parse_double(optarg, 0.0, 1, 86400.0, &o->timeout);
814 break;
815 case OPT_ROUNDS:
816 ok = parse_int(optarg, 0, 1000000, &o->rounds);
817 break;
818 case OPT_FACTOR:
819 ok = parse_double(optarg, 1.0, 1, 1e6, &o->factor);
820 break;
821 case OPT_PAUSE:
822 ok = parse_double(optarg, 0.0, 0, 86400.0, &o->pause);
823 break;
824 case OPT_MACHINE:
825 o->machine = 1;
826 break;
827 case OPT_HELP:
828 if (rank == 0) {
829 usage(stdout, prog);
830 }
831 return -1;
832 }
833 if (!ok) {
834 if (rank == 0) {
835 fprintf(stderr, "%s: invalid value '%s' for --%s: expected %s\n", prog, optarg, option_table[k].name, option_table[k].expect);
836 }
837 return usage_hint(prog);
838 }
839 }
840 if (optind < argc) {
841 if (rank == 0) {
842 fprintf(stderr, "%s: unexpected argument '%s'\n", prog, argv[optind]);
843 }
844 return usage_hint(prog);
845 }
846 return 0;
847}

863 /* The default handler would abort the job on the first transport error, before anything is reported. */
864 MPI_Comm_set_errhandler(MPI_COMM_WORLD, MPI_ERRORS_RETURN);

899 * each round at w bounds every rank's posting to ~timeout after its start,
900 * so 2*timeout cannot flag a working link merely because its peer was
901 * delayed by failures elsewhere. */
902 w = size > 1 ? timeout / (size - 1) : 0.0;
903 for (k = 1; k < size; ++k) {
904 int src = (rank - k + size) % size, dst = (rank + k) % size;

910 progress();
911 }

914 progress();
915 }

918 if (rstate[j] == DONE && rbuf[j] != j) {
919 rstate[j] = FAILED;
920 }
921 }

923 /* Only a completed receive proves a link: eager sends may complete without delivery. */
924 nfail = count_not_done(rstate, size);
925 mine.nfail = nfail;

928 mine.range[mine.nranges][0] = lo;

951 printf("-");
952 } else {
953 print_runs(rstate, size);
954 }

959 if (rank != 0) {
960 char go = PENDING;
961 int gobuf;
962
963 post(1, &mine, (int)sizeof mine, MPI_BYTE, 0, TAG_REPORT, &rep);
964 while (nact > 0 && MPI_Wtime() - t0 < 3 * timeout) {
965 progress();
966 }

969 fflush(stdout);

972 /* Finalizing with requests pending is erroneous. Rank 0 normally
973 * aborts the job first; this is the fallback if it can't. */
974 while (MPI_Wtime() - t0 < 4 * timeout) {
975 progress();
976 }
977 MPI_Abort(MPI_COMM_WORLD, 1);
978 }
979 if (rounds == 0) {
980 MPI_Finalize();
981 return 0;
982 }
983 /* Only rank 0 knows whether every link works; latency rounds must not start otherwise. */
984 post(0, &gobuf, 1, MPI_INT, 0, TAG_GO, &go);
985 while (nact > 0 && MPI_Wtime() - t0 < 4 * timeout) {
986 progress();
987 }
988 if (go != DONE) {
989 printf("rank %d (%s): no go-ahead from rank 0 for latency rounds\n", rank, mine.host);
990 fflush(stdout);
991 MPI_Abort(MPI_COMM_WORLD, 1);
992 }

1003 MPI_Abort(MPI_COMM_WORLD, 1);

1006 repstate[0] = DONE;

1012 }

1018

1020 continue;
1021 }

1026 unheard[i] = 1;

1028 }

1032 }

1037 if (k == 0 || strcmp(all[order[k]].host, all[order[k - 1]].host) != 0) {
1038 ++nhosts;
1039 }

1057 printf(" no report, host unknown, from rank(s): ");
1058 print_runs(repstate, size);
1059 printf("\n");
1060 }
1061 if (ntrunc > 0) {

1065 MPI_Abort(MPI_COMM_WORLD, 1);
1066 }

1068 printf("connectivity OK: each of %d rank(s) on %d host(s) received from every other rank (%.2f s)\n", size, nhosts, elapsed);
1069 fflush(stdout);
1070 if (rounds == 0) {
1071 MPI_Finalize();
1072 return 0;
1073 }

1076 fprintf(stderr, "rank 0: out of memory\n");
1077 MPI_Abort(MPI_COMM_WORLD, 1);
1078 }
1079 for (j = 1; j < size; ++j) {
1080 post(1, &rank, 1, MPI_INT, j, TAG_GO, &gostate[j]);

1110

1124 while (MPI_Wtime() - tl_end < 5 * timeout) {
1125 progress();
1126 }

1128 } else {
1129 long nps = (long)size * (size - 1) / 2, off = nmine, nuse = 0, nev = 0;
1130 pairstat_t *ps = (pairstat_t *)calloc((size_t)nps + 1, sizeof *ps), *use = (pairstat_t *)calloc((size_t)nps + 1, sizeof *use);
1131 event_t *evall = (event_t *)calloc((size_t)size * MAXEV, sizeof *evall), *evuse = (event_t *)calloc((size_t)size * MAXEV, sizeof *evuse);
1132 int *hdrs = (int *)calloc(3 * (size_t)size, sizeof *hdrs), nmiss = 0, nstall = 0, ngave = 0, ntrunc = 0, outliers;
1133 char *hst = (char *)calloc(size, 1), *est = (char *)calloc(size, 1), *sst = (char *)calloc(size, 1);
1134

1136 fprintf(stderr, "rank 0: out of memory for latency statistics\n");
1137 MPI_Abort(MPI_COMM_WORLD, 1);
