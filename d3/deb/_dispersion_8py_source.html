417 # L mode
418 def kL(w):
419 return 1./SI.c * np.sqrt(w**2 - wpe**2/(1.+Wce/w) - wpi**2/(1.-Wci/w))
420 # lf L mode
421 maskLlf = np.logical_and(w>0, w<Wci)
422 wleftlf = w[maskLlf]
423 kleftlf = kL(wleftlf)
424 # hf L mode
425 wLcutoff = 0.5*(np.sqrt((Wci+Wce)**2+4.*wpe**2+4.*wpi**2)-(Wce-Wci))
426 maskLhf = w>wLcutoff
427 wlefthf = w[maskLhf]
428 klefthf = kL(wlefthf)

430 # R mode
431 def kR(w):
432 return 1./SI.c * np.sqrt(w**2 - wpe**2/(1.-Wce/w) - wpi**2/(1.+Wci/w))
433 # lf R mode
434 maskRlf = np.logical_and(w>0, w<Wce)
435 wrightlf = w[maskRlf]
436 krightlf = kR(wrightlf)
437 # hf R mode
438 wRcutoff = 0.5*(np.sqrt((Wci+Wce)**2+4.*wpe**2+4.*wpi**2)+(Wce-Wci))
439 maskRhf = w>wRcutoff
440 wrighthf = w[maskRhf]
441 krighthf = kR(wrighthf)

443 # Langmuir mode
