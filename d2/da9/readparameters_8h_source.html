60 template <typename T>
61 static CLI::Option* add(const std::string& name, const std::string& desc, T& var,
62 std::optional<T> defval = std::nullopt) {
63 int rank;
64 MPI_Comm_rank(MPI_COMM_WORLD, &rank);
65 CLI::Option* opt;
66 std::vector<std::string> cmdNameVec = parseCommandName(name);
67 if (cmdNameVec.size() == 2) {
68 auto subcom = cmdNameVec.at(0);
69 auto namein = cmdNameVec.at(1);
70 CLI::App* sub = nullptr;
71
72 if (!isSubComParsed[subcom]) {
73 if (subcommandDescriptions.find(subcom)!=subcommandDescriptions.end()) {
74 sub = app->add_subcommand(subcom, subcommandDescriptions[subcom]);
75 } else {
76 sub = app->add_subcommand(subcom, "");
77 }
78 } else {
79 sub = app->get_subcommand(subcom);
80 };
81 if (sub != nullptr) {
82 std::string dashes = "-";
83 if (namein.size() != 1) {
84 dashes += "-";
85 }
86 opt =
87 sub->add_option((dashes + namein).c_str(), var, desc.c_str())->capture_default_str();
88 isSubComParsed[subcom] = true;
89 } else {
90 std::cerr << "Something went wrong with adding subcommand " + subcom + "!" << std::endl;
91 abort();
92 };
93 } else {
94 opt = app->add_option(("--" + cmdNameVec.at(0)).c_str(), var, desc.c_str())
95 ->capture_default_str();
96 }
97 if (defval && opt != nullptr) {
98 std::stringstream ss;
99 static constexpr bool h = (std::is_same<std::remove_cv<std::remove_reference<T>>, std::string>::value);
100 static constexpr bool isVec = (is_vector<T>{});
101 static constexpr bool n = (std::is_floating_point<T>::value);
102 // fixes issue with default val precision BUT doesnt fix it for vectors for currently.
103 // however it only applies if the the default value was not given as a variable rather than a literal (if even
104 // possible)
105 if constexpr (n && !h && !isVec) {
106 ss << std::setprecision(std::numeric_limits<double>::digits10 + 1) << *defval;
107 opt->default_val(ss.str());
108 } else {
109 opt->default_val(*defval);
110 }
111 }
112
113 return opt;
114 }
115 static void parseComposing() {
116 int rank;
117 std::ifstream configFile;
118 std::string line;
119 std::string subcom;
120 std::string commandName;
121 bool hasConfig = true;
122 MPI_Comm_rank(MPI_COMM_WORLD, &rank);
123 if (rank == MASTER_RANK && app->get_config_ptr()->count() == 0) {
124 hasConfig = false;
125 }
126 MPI_Bcast(&hasConfig, sizeof(bool), MPI_BYTE, MASTER_RANK, MPI_COMM_WORLD);
127 if (!hasConfig) {
128 return;
129 }
130 if (rank == MASTER_RANK) {
131 configFile.open(app->get_config_ptr()->as<std::string>());
132 if (configFile.is_open()) {
133 while (getline(configFile, line)) {
134 // remove white spaces
135 auto it = std::remove_if(line.begin(), line.end(), [](unsigned char x) { return std::isspace(x); });
136 line.erase(it, line.end());
137
138 // skip if empty
139 if (line.empty()) {
140 continue;
141 }
142 // check if subcommand, if found make it current subcommand to be parsed
143 if (line.front() == '[') {
144 if (line.back() != ']') {
145 std::cerr << "Invalid configuration line, found a line starting with '[' but it did not end with "
146 "']':\n" +
147 line
148 << std::endl;
149 }
150 subcom = line.substr(1, line.length() - 2);
151 continue;
152 }
153 // Find '=' to capture option name and value as string_view for now
154 // Find eqIndx, skip if not found
155 auto eqIndx = line.find('=');
156 if (eqIndx == std::string::npos) {
157 continue;
158 }
159 std::string_view optName = std::string_view(line).substr(0, eqIndx);
160 std::string_view optValue = std::string_view(line).substr(eqIndx + 1, line.size());
161
162 // Are we inside subcommand or not?
163 if (!subcom.empty()) {
164 commandName = subcom + '.' + std::string(optName);
165 } else {
166 commandName = std::string(optName);
167 }
168
169 // Skip vector values and let CLI11 handle them
170 if (optValue.front() == '[') {
171 // Do we need to do something?
172 isOptionParsed.erase(commandName);
173 continue;
174 }
175
176 // Skip if the command is not addComposing/not known
177 if (auto search = optionsComposing.find(commandName); search == optionsComposing.end()) {
178 continue;
179 }
180 // this is used for checking whether to pass the handling to CLI11 later(or rather whether to rerun
181 // callback)
182 if (auto search = isOptionParsed.find(commandName); search == isOptionParsed.end()) {
183 continue;
184 }
185
186 if (optionsComposing[commandName].empty()) {
187 optionsComposing[commandName] = std::string(optValue);
188 } else {
189 optionsComposing[commandName] = optionsComposing[commandName] + ',' + std::string(optValue);
190 }
191 isOptionParsed[commandName] = false;
192 }
193 configFile.close();
194 }
195 }
196 // Add brackets and throw the values to CLI option
197 for (std::map<std::string, std::string>::iterator iter = optionsComposing.begin(); iter != optionsComposing.end();
198 ++iter) {
199 std::string commandName = iter->first;
200 if (rank == MASTER_RANK) {
201 if (auto search = isOptionParsed.find(commandName); search == isOptionParsed.end()) {
202 isOptionParsed[commandName] = true;
203 } else {
204 isOptionParsed[commandName] = false;
205 }
206 }
207 //Parse the values and run CLI11 callback
208 MPI_Bcast(&isOptionParsed[commandName], sizeof(bool), MPI_BYTE, MASTER_RANK, MPI_COMM_WORLD);
209 if (!isOptionParsed[commandName]) {
210 iter->second = '[' + iter->second + ']';
211 std::string optVal = iter->second;
212 int strSize = optVal.size();
213 MPI_Bcast(&strSize, 1, MPI_INT, MASTER_RANK, MPI_COMM_WORLD);
214
215 if (rank != MASTER_RANK) {
216 optVal.resize(strSize);
217 }
218 MPI_Bcast(optVal.data(), strSize, MPI_CHAR, MASTER_RANK, MPI_COMM_WORLD);
219
220 auto opt = getOption(commandName);
221
222 int n = std::count(optVal.begin(), optVal.end(), ',');
223 bool isValid = true;
224
225 if (optVal != "[]" && n == 0)
226 n = 1; // This means we have one argument
227
228 if (opt->get_items_expected() == 0 && optVal == "[]") {
229 continue;
230 } else if ((opt->get_items_expected_max() < n || opt->get_items_expected_min() > n)) {
231 std::cerr << "Option " << commandName << " expected " << opt->get_items_expected()
232 << " argument(s), but found " << n << "." << std::endl;
233 isValid = false;
234 }
235 if (!isValid) {
236 abort();
237 }
238
239 opt->clear();
240 opt->add_result(optVal);
241 opt->run_callback();
242 }
243 }
244 }

303 static CLI::Option* getOption(const std::string& name) {
304 if (auto indx = name.find('.'); indx != std::string::npos) {
305 std::string subcomName = name.substr(0, indx);
306 std::string optName = name.substr(indx + 1, name.size());
307 CLI::App* subcom = app->get_subcommand(subcomName);
308 std::string dashes = "-";
309 if (optName.size() != 1) {
310 dashes += "-";
311 }
312 return subcom->get_option(dashes + optName);
313 }
314
315 std::string dashes = "";
316 if (name.front() != '-') {
317 dashes = "-";
318 if (name.size() != 1) {
319 dashes += "-";
320 }
321 }
322 return app->get_option(dashes + name);
323 }

328 static bool removeOption(const std::string& name) {
329 if (auto indx = name.find('.'); indx != std::string::npos) {
330 std::string subcomName = name.substr(0, indx);
331 std::string optName = name.substr(indx + 1, name.size());
332 CLI::App* subcom = app->get_subcommand(subcomName);
333 subcom->remove_option(subcom->get_option(optName));
334 } else {
335 app->remove_option(app->get_option(name));
336 }
337 return true;
