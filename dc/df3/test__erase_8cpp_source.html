16template <typename GID>
17static bool consistent(const OpenBucketHashtable<GID, uint32_t>& h, const std::unordered_map<GID, uint32_t>& ref) {
18 if (h.size() != ref.size()) {
19 return false;
20 }
21 for (const auto& [key, val] : ref) {
22 const auto it = h.find(key);
23 if (it == h.end() || it->second != val) {
24 return false;
25 }
26 }
27 size_t n = 0;
28 for (auto it = h.begin(); it != h.end(); ++it, ++n) {
29 if (ref.count(it->first) == 0) {
30 return false;
31 }
32 }
33 return n == ref.size();
34}

36// Home bucket of a key in a fresh (16-bucket) table.
37static size_t homeOf(uint32_t k) {
38 OpenBucketHashtable<uint32_t, uint32_t> t;
39 t.at(k) = 0;
40 return t.begin().getIndex();
41}

43// Smallest key >= start, not in used, whose home bucket is target.
44static uint32_t keyWithHome(size_t target, uint32_t start, const std::unordered_map<uint32_t, uint32_t>& used) {
45 uint32_t k = start;
46 while (homeOf(k) != target || used.count(k)) {
47 ++k;
48 }
49 return k;
50}

52// Two keys sharing a home bucket in a fresh table: erasing the first must keep the second findable.
53static bool collidingPair() {
54 const uint32_t a = 1;
55 const uint32_t b = keyWithHome(homeOf(a), a + 1, {});
56 OpenBucketHashtable<uint32_t, uint32_t> h;
57 std::unordered_map<uint32_t, uint32_t> ref;
58 h.at(a) = 10; ref[a] = 10;
59 h.at(b) = 20; ref[b] = 20;
60 h.erase(a); ref.erase(a);
61 return consistent(h, ref);
62}

64// Nearly full table whose cluster spans the whole table: buckets 0..13 hold keys at home, bucket 14
65// holds a second key with home 13. Erasing bucket 0 must not move that key back across the table
66// into bucket 0, which would leave an empty bucket 14 between its home and it.
67static bool wrappedCluster() {
68 OpenBucketHashtable<uint32_t, uint32_t> h;
69 std::unordered_map<uint32_t, uint32_t> ref;
70 for (size_t bucket = 0; bucket <= 14; ++bucket) {
71 const uint32_t k = keyWithHome(bucket == 14 ? 13 : bucket, 1, ref);
72 h.at(k) = bucket;
73 ref[k] = bucket;
74 }
75 if (h.bucket_count() != 16) {
76 return false; // layout assumption broken, not the bug under test
77 }
78 const auto first = h.begin();
79 ref.erase(first->first);
80 h.erase(first);
81 return consistent(h, ref);
82}

84// Random interleaved inserts and erases. Small maxKeys keeps 16-bucket tables nearly full,
85// where clusters wrap around the table.
86template <typename GID>
87static long randomOps(uint32_t maxKeys, int runs, uint32_t seed) {
88 std::mt19937 rng(seed);
89 std::uniform_int_distribution<GID> gid(0, 4 * maxKeys + 50);
90 long failingRuns = 0;
91 for (int run = 0; run < runs; ++run) {
92 OpenBucketHashtable<GID, uint32_t> h;
93 std::unordered_map<GID, uint32_t> ref;
94 for (uint32_t step = 0; step < 4 * maxKeys; ++step) {
95 if ((rng() & 1) && ref.size() < maxKeys) {
96 const GID k = gid(rng);
97 h.at(k) = step;
98 ref[k] = step;
99 } else if (!ref.empty()) {
100 auto it = ref.begin();
101 std::advance(it, rng() % ref.size());
102 h.erase(it->first);
103 ref.erase(it);
104 }
105 if (!consistent(h, ref)) {
106 ++failingRuns;
107 break;
108 }
109 }
110 }
111 return failingRuns;
112}
