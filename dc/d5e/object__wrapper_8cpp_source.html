155 return false;
156 }
157

159 species.charge *= physicalconstants::CHARGE;
160 if (species.mass_units == "PROTON") {

162 } else if (species.mass_units == "ELECTRON") {
163 species.mass *= physicalconstants::MASS_ELECTRON;

170 if (vMesh.gridLength[0] > MAX_BLOCKS_PER_DIM || vMesh.gridLength[1] > MAX_BLOCKS_PER_DIM ||
171 vMesh.gridLength[2] > MAX_BLOCKS_PER_DIM) {
172
173 // Build error message as a string first, so that the cerr output hapens atomically and we don't spam
174 // thousands of unreadable lines through each other
175 species.mass = 1.5;
176 std::string errormsg = "(VSPACE) ERROR: Velocity mesh for population " + species.name +

181 std::cerr << errormsg;
182 }
183 }
184
185 /* Special handling of WID=8; halve the number of blocks */
186 int myRank;
187 MPI_Comm_rank(MPI_COMM_WORLD,&myRank);
188 if ((WID==8 && P::adaptGPUWID)) {
189 // First verify that we can halve the value2
190 if ((vMesh.gridLength[0]%2==0) && (vMesh.gridLength[1]%2==0) && (vMesh.gridLength[2]%2==0)) {
191 vMesh.gridLength[0] /= 2;
192 vMesh.gridLength[1] /= 2;
193 vMesh.gridLength[2] /= 2;
194 if(myRank==MASTER_RANK) {
195 std::cerr<<" Note: Using WID=8; Halving velocity block counts per dimension. Deactivate with parameter adaptGPUWID=false."<<std::endl;
196 }
197 } else {
198 if(myRank==MASTER_RANK) {
199 std::cerr<<" Warning: Using WID=8 but odd number of velocity blocks! Cannot halve the blocks count."<<std::endl;
200 }
201 }
202 }
203
204 vMesh.blockLength[0] = vMesh.blockLength[1] = vMesh.blockLength[2] = WID;
205
206 const Real EPSILON = 1.e-25;

208 // Energy stored internally in SI units
209 species.SolarWindEnergy = 0.5 * species.mass * species.SolarWindSpeed * species.SolarWindSpeed;
210 } else {
211 species.SolarWindEnergy = species.SolarWindEnergy*physicalconstants::CHARGE;

213 // Convert from eV to SI units
214 species.precipitationEmin = species.precipitationEmin*physicalconstants::CHARGE;
215 species.precipitationEmax = species.precipitationEmax*physicalconstants::CHARGE;
216
217 getObjectWrapper().particleSpecies.resize(getObjectWrapper().particleSpeciesRead.size());
218 getObjectWrapper().particleSpecies.at(i)=species;
219 }
