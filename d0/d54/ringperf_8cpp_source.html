201 double t;
202 int i;
203
204 memset(b, 0, sizeof b);
205 /* Untimed first exchange: the partner may still be busy with its previous link. */
206 MPI_Send(b, LATBYTES, MPI_BYTE, peer, TAG_LAT, MPI_COMM_WORLD);
207 MPI_Recv(b, LATBYTES, MPI_BYTE, peer, TAG_LAT, MPI_COMM_WORLD, MPI_STATUS_IGNORE);
208 for (i = 0; i < opt.pingpongs; ++i) {
209 t = MPI_Wtime();

238 double t = MPI_Wtime(), sent = (double)opt.window * opt.size;
239 int i, ack;

242 MPI_Isend(sbuf, opt.size, MPI_BYTE, peer, TAG_BW, MPI_COMM_WORLD, &req[i]);
243 }
244 MPI_Waitall(opt.window, req, MPI_STATUSES_IGNORE);
245 /* Local completion only means the data left the buffer; the acknowledgement means it arrived. */
246 MPI_Recv(&ack, 1, MPI_INT, peer, TAG_ACK, MPI_COMM_WORLD, MPI_STATUS_IGNORE);
247 t = MPI_Wtime() - t;
248 if (record && t > 0) {

250 *time += t;

268static void one_round(stats_t *st, int record) {
269 int own = rank < nlinks() ? rank : -1, left = (rank - 1 + nranks) % nranks, g;
270
271 for (g = 0; g < ngroups(); ++g) {
272 if (own >= 0 && link_group(own) == g) {
273 /* The left end of a link times latency and the forward direction, the right end the way back. */
274 lat_initiate((rank + 1) % nranks, st, record);
275 bw_send((rank + 1) % nranks, &st->fwd_bytes, &st->fwd_time, &st->fwd_worst, record);
276 bw_recv((rank + 1) % nranks);
277 } else if (left < nlinks() && left != own && link_group(left) == g) {
278 lat_answer(left);
279 bw_recv(left);
280 bw_send(left, &st->back_bytes, &st->back_time, &st->back_worst, record);
281 }
282 }
