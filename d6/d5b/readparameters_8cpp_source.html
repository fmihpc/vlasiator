95void Readparameters::applyAssignment(Option& opt, const std::string& rawValue, std::set<std::string>& touched) {
96 const bool first = touched.insert(opt.name).second;
97 if (first) {
98 opt.clearValue();
99 }
100 opt.wasSet = true;
101
102 if (rawValue.size() >= 2 && rawValue.front() == '[' && rawValue.back() == ']') {
103 const std::string inner = rawValue.substr(1, rawValue.size() - 2);
104 if (!inner.empty()) {
105 QdArgParser<','> parser(inner.data(), inner.size());
106 for (auto it = parser.begin(); it != parser.end(); ++it) {
107 const std::string tok(*it);
108 if (!tok.empty()) {
109 opt.assignOne(tok);
110 }
111 }
112 }
113 } else {
114 opt.assignOne(rawValue);
115 }
116}

118void Readparameters::applyArgTokens(const std::vector<std::string>& tokens, bool extras, std::vector<std::string>& invalid, std::vector<std::string>& filenames) {
119 std::set<std::string> touched;
120 std::size_t i = 0;
121 while (i < tokens.size()) {
122 const std::string& tok = tokens[i];
123 if (tok.size() < 3 || tok[0] != '-' || tok[1] != '-') {
124 ++i;
125 filenames.push_back(tok);
126 continue;
127 }
128 const std::string body = tok.substr(2);
129 std::string name = body;
130 std::string value;
131 bool hasValue = false;
132 if (const auto eq = body.find('='); eq != std::string::npos) {
133 name = body.substr(0, eq);
134 value = body.substr(eq + 1);
135 hasValue = true;
136 }
137
138 const std::string key = normalizeName(name);
139 auto it = registry().find(key);
140 if (it == registry().end()) {
141 if (!extras) {
142 invalid.push_back(name);
143 }
144 ++i;
145 continue;
146 }
147 Option& opt = it->second;
148
149 if (opt.isFlag) {
150 applyAssignment(opt, hasValue ? value : std::string(""), touched);
151 ++i;
152 continue;
153 }
154
155 if (!hasValue) {
156 if (i + 1 < tokens.size()) {
157 value = tokens[i + 1];
158 i += 2;
159 } else {
160 invalid.push_back(name + " (missing value)");
161 ++i;
162 continue;
163 }
164 } else {
165 ++i;
166 }
167 applyAssignment(opt, value, touched);
168 }
169}

171void Readparameters::applyConfigFile(const std::string& filename, bool extras, std::vector<std::string>& invalid) {
172 std::FILE* fp = std::fopen(filename.c_str(), "rb");
173 if (!fp) {
174 return;
175 }
176 std::fseek(fp, 0, SEEK_END);
177 const long fsize = std::ftell(fp);
178 std::fseek(fp, 0, SEEK_SET);
179 std::string buffer;
180 if (fsize > 0) {
181 buffer.resize(static_cast<std::size_t>(fsize));
182 buffer.resize(std::fread(buffer.data(), 1, buffer.size(), fp));
183 }
184 std::fclose(fp);
185 if (buffer.empty()) {
186 return;
187 }
188
189 std::string section;
190 std::set<std::string> touched;
191 QdArgParser<'\n'> parser(buffer.data(), buffer.size());
192 for (auto lineIt = parser.begin(); lineIt != parser.end(); ++lineIt) {
193 std::string rawLine(*lineIt);
194 //allow # comments inline
195 const auto commentPos = rawLine.find('#');
196 if (commentPos != std::string::npos) {
197 rawLine.erase(commentPos);
198 }
199
200 const auto firstNonSpace = rawLine.find_first_not_of(" \t\r\n");
201 if (firstNonSpace == std::string::npos) {
202 continue;
203 }
204 std::string line = rawLine;
205 line.erase(std::remove_if(line.begin(), line.end(), [](unsigned char ch) { return std::isspace(ch); }), line.end());
206 if (line.empty()) {
207 continue;
208 }
209
210 if (line.front() == '[') {
211 if (line.back() == ']') {
212 section = line.substr(1, line.size() - 2);
213 } else {
214 std::cerr << "Invalid configuration line, found a line starting with '[' which does not end with ']':\n"
215 << rawLine << std::endl;
216 }
217 continue;
218 }
219
220 const auto eq = line.find('=');
221 if (eq == std::string::npos) {
222 continue;
223 }
224 const std::string name = line.substr(0, eq);
225 const std::string value = line.substr(eq + 1);
226 const std::string fullName = section.empty() ? name : section + '.' + name;
227 const std::string key = normalizeName(fullName);
228
229 auto it = registry().find(key);
230 if (it == registry().end()) {
231 if (!extras) {
232 invalid.push_back(fullName);
233 }
234 continue;
235 }
236 applyAssignment(it->second, value, touched);
237 }
238}

240std::string Readparameters::serializeAll() {
241 std::ostringstream oss;
242 bool first = true;
243 for (const auto& key : registryOrder()) {
244 const Option& opt = registry().at(key);
245 if (!opt.wasSet) {
246 continue;
247 }
248 if (!first) {
249 oss << ' ';
250 }
251 first = false;
252 oss << "--" << key << '=' << opt.serializeValue();
253 }
254 return oss.str();
255}

257std::string Readparameters::finalizeFileName(const std::vector<std::string>& tokens) {
258 static const std::string prefix = "--run_config=";
259 for (std::size_t i = 0; i < tokens.size(); ++i) {
260 const std::string& tok = tokens[i];
261 if (tok.rfind(prefix, 0) == 0) {
262 return tok.substr(prefix.size());
263 }
264 if (tok == "--run_config" && i + 1 < tokens.size()) {
265 return tokens[i + 1];
266 }
267 }
268 return configFileName;
269}

271void Readparameters::helpMessage() {
272 if (!helpRequested) {
273 return;
274 }
275 int rank;
276 MPI_Comm_rank(MPI_COMM_WORLD, &rank);
277 if (rank == MASTER_RANK) {
278 cout << "Usage: main [options (options given on the command line override options given "
279 "everywhere else)], where options are:\n"
280 << endl;
281
282 map<string, vector<string>> bySection;
283 for (const auto& key : registryOrder()) {
284 const auto dot = key.find('.');
285 const string section = (dot == string::npos) ? string() : key.substr(0, dot);
286 bySection[section].push_back(key);
287 }
288
289 auto printGroup = [](const string& section, const vector<string>& keys) {
290 if (section.empty()) {
291 cout << "General options:" << endl;
292 } else {
293 auto it = subcommandDescriptions.find(section);
294 cout << section;
295 if (it != subcommandDescriptions.end() && !it->second.empty()) {
296 cout << ": " << it->second;
297 }
298 cout << ":" << endl;
299 }
300 for (const auto& key : keys) {
301 const Option& opt = registry().at(key);
302 cout << " --" << key;
303 if (!opt.isFlag) {
304 cout << "=<" << (opt.isVector ? "list" : "value") << ">";
305 }
306 cout << "\n " << opt.desc;
307 if (!opt.isFlag) {
308 cout << " (default: " << opt.defaultStr << ")";
309 }
310 cout << endl;
311 }
312 cout << endl;
313 };
314
315 if (const auto it = bySection.find(string()); it != bySection.end()) {
316 printGroup(string(), it->second);
317 }
318 for (const auto& entry : bySection) {
319 if (entry.first.empty()) {
320 continue;
321 }
322 printGroup(entry.first, entry.second);
323 }
324 }
325 MPI_Finalize();
326 exit(0);
327}

346void Readparameters::parse(std::vector<std::string>& invalid, std::vector<std::string>& filenames, bool extras) {
347 int rank;
348 MPI_Comm_rank(MPI_COMM_WORLD, &rank);
349 std::string conf;
350 int confsize = 0;
351
352 if (rank == MASTER_RANK) {
353 resetAll();
354 const std::vector<std::string> tokens = make_tokens(argc-1, argv+1); // skip the program name in argv[0]
355 configFileName = finalizeFileName(tokens);
356
357 applyConfigFile(configFileName, extras, invalid);
358 applyArgTokens(tokens, extras, invalid, filenames);
359 if (!extras && !invalid.empty()) {
360 std::cerr << "Error parsing config, following options are invalid:\n";
361 for (const auto& name : invalid) {
362 std::cerr << " " << name << "\n";
363 }
364 std::cerr << std::endl;
365 MPI_Finalize();
366 exit(1);
367 }
368
369 conf = serializeAll();
370 confsize = static_cast<int>(conf.size());
371 }
372
373 MPI_Bcast(&confsize, 1, MPI_INT, MASTER_RANK, MPI_COMM_WORLD);
374 if (rank != MASTER_RANK) {
375 resetAll();
376 conf.resize(confsize);
377 }
378 MPI_Bcast(conf.data(), confsize, MPI_CHAR, MASTER_RANK, MPI_COMM_WORLD);
379
380 if (rank != MASTER_RANK) {
381 applyArgTokens(make_tokens(conf), true, invalid, filenames);
382 }
